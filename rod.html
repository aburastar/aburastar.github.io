  <script src="https://cdnjs.cloudflare.com/ajax/libs/three.js/r128/three.min.js"></script>

  <script>

    // Set up scene, camera, renderer

    const scene = new THREE.Scene();

    const camera = new THREE.PerspectiveCamera(75, window.innerWidth / window.innerHeight, 0.1, 1000);

    camera.position.set(0.5, 0.5, 0.5);

    camera.lookAt(0, 0, 0);

    const renderer = new THREE.WebGLRenderer();

    renderer.setSize(window.innerWidth, window.innerHeight);

    document.body.appendChild(renderer.domElement);


    // Add lighting

    const light = new THREE.DirectionalLight(0xffffff, 0.8);

    light.position.set(1, 1, 1);

    scene.add(light);

    scene.add(new THREE.AmbientLight(0x404040));


    // Your provided code

    class RodCurve extends THREE.Curve {

      constructor(L=1.0, A=0.05){ super(); this.L=L; this.A=A; }

      getPoint(t){

        const x = this.A * Math.sin(Math.PI * t);

        const y = 0.0;

        const z = this.L * t;

        return new THREE.Vector3(x,y,z);

      }

    }


    function radiusAt(t, r0=0.015, h=0.008, th=0.7, sigma=0.12, alpha=0.003){

      const gauss = Math.exp(-((t-th)*(t-th))/(2*sigma*sigma));

      return Math.max(0.001, r0 + h*gauss - alpha*t);

    }


    const path = new RodCurve(0.22, 0.02);

    const tubularSeg = 300, radialSeg = 24;

    const baseR = 0.02;

    const geom = new THREE.TubeGeometry(path, tubularSeg, baseR, radialSeg, false);


    const pos = geom.attributes.position;

    for (let i=0; i<pos.count; i++){

      const ring = Math.floor(i / (radialSeg+1));

      const t = ring / tubularSeg;

      const wanted = radiusAt(t);

      const scale = wanted / baseR;

      const v = new THREE.Vector3().fromBufferAttribute(pos, i);

      const c = path.getPoint(t);

      v.sub(c).multiplyScalar(scale).add(c);

      pos.setXYZ(i, v.x, v.y, v.z);

    }

    pos.needsUpdate = true;


    const mat = new THREE.MeshStandardMaterial({ roughness:0.35, metalness:0.0 });

    const rod = new THREE.Mesh(geom, mat);

    scene.add(rod);


    // Animation loop

    function animate() {

      requestAnimationFrame(animate);

      rod.rotation.y += 0.01; // Rotate for better viewing

      renderer.render(scene, camera);

    }

    animate();


    // Handle window resize

    window.addEventListener('resize', () => {

      camera.aspect = window.innerWidth / window.innerHeight;

      camera.updateProjectionMatrix();

      renderer.setSize(window.innerWidth, window.innerHeight);

    });

  </script>
